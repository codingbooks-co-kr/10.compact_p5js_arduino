// 배열을 사용하여 세 개의 원을 오른쪽으로 이동 (→배열과 반복문을 함께 사용할 것을 추천)

let x = [-50, 0, 50];		// 배열의 선언 및 초기화를 한 번에 처리

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  ellipse(x[0], 80, 50, 50);
  ellipse(x[1], 160, 50, 50);
  ellipse(x[2], 240, 50, 50);
  x[0]++;
  x[1]++;
  x[2]++;
}
