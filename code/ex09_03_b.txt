// p5.js: 조도센서값 수신하여 사각형 이동

let port = new p5.WebSerial();
let value = 0;

function setup( ) {
  createCanvas(300, 500);
  port.getPorts();
  port.on('noport', makePortButton);
  port.on('portavailable', openPort);
  port.on('data', serialEvent);
}

function draw() {
  background(0, 10);	// 페이딩 효과
  noStroke();		// 외곽선 제거
  fill(255, 255, 0);
  let h = int(map(value, 0, 255, 0, height));
  rect(50, h, 200, 20);
}

function serialEvent() {
  value = port.read();	// 센서값 수신
}

function makePortButton() {
  let portButton = createButton('포트선택');
  portButton.mousePressed(choosePort);
  function choosePort() {
    port.requestPort();
  }
}

function openPort() {
  port.open().then(initiateSerial);
  function initiateSerial() {
    console.log('포트 오픈');
  }
}
