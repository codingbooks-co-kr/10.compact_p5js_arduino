// 가로선 그리기

let x1 = 50.5, x2 = 250.5, y = 100.5;	// 실수로 초기화

function setup() {
  createCanvas(300, 300);
  background(200);

  line(x1, y%30, x2, y%30);			// 나눈 나머지
  line(x1, y/2, x2, y/2);			// 나눗셈
  line(x1, y-30, x2, y-30);			// 뺄셈
  line(x1, y, x2, y);
  line(x1, y+30, x2, y+30);			// 덧셈
  line(x1, y*2, x2, y*2);			// 곱셈
}
